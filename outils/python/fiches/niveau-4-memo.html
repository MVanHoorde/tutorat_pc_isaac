Exemple commenté 1 · l'état final d'un acide faible avec bisect

import numpy as np
from scipy.optimize import bisect      # 14 Mo téléchargés la 1re fois

Ka = 10 ** (-4.8)                       # acide éthanoïque
C = 1.0e-2                              # concentration apportée, en mol/L

def f(x):
    """Écart à l'équilibre ; x = [H3O+] en mol/L, 0 <= x < C."""   # la spécification
    return x**2 / (C - x) - Ka

assert f(0) < 0 and f(0.001) > 0          # on vérifie le changement de signe AVANT
x_f = bisect(f, 0, 0.001)              # f sans parenthèses : on passe la fonction
print('tau =', round(x_f / C, 4), ' pH =', round(-np.log10(x_f), 2))

tau = 0.039 pH = 3.41

Ne pas prendre b = C : C − x s'annulerait (division par zéro).

Exemple commenté 2 · valider une droite, puis l'incertitude sur sa pente

import numpy as np
import matplotlib.pyplot as plt

c = np.array([1.0, 2.5, 5.0, 8.0, 10.0])                # TP4, en mmol/L
sigma = np.array([0.162, 0.330, 0.605, 0.925, 1.158])   # en mS/cm
u_sigma = 0.015                                         # incertitude-type sur chaque σ

# 1. régression, puis écart normalisé de chaque point au modèle
a, b = np.polyfit(c, sigma, 1)
E = (sigma - (a * c + b)) / u_sigma     # tous |E| < 2 et signes mêlés : modèle validé
print('E =', np.round(E, 2))                # np.round arrondit tout un tableau

# 2. Monte-Carlo : on refait la régression sur 1000 jeux de mesures simulés
np.random.seed(1)                       # fixe le hasard : mêmes tirages à chaque exécution
pentes = []
for i in range(1000):
    sigma_sim = sigma + np.random.normal(0, u_sigma, 5)   # chaque point bouge au hasard
    p, o = np.polyfit(c, sigma_sim, 1)
    pentes.append(p)                    # on garde la pente de ce tirage
print('a =', round(a, 4), ' u(a) =', round(np.std(pentes, ddof=1), 4))

# 3. le graphique « raisonné » : barres d'incertitude + modèle
plt.errorbar(c, sigma, yerr=u_sigma, fmt='o', label='mesures')
plt.plot(c, a * c + b, 'r-', label='modèle affine')
plt.legend()
plt.show()

E = [-0.07 0.13 0.13 -0.53 0.33] a = 0.11 u(a) = 0.0019

Résultat : a = (0,110 ± 0,002) mS·cm⁻¹ par mmol·L⁻¹. Python écrit 0.11 : il n'affiche pas les zéros de fin.

Exemple commenté 3 · un pendule avec odeint

import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import odeint

g = 9.81
L = 1.0

def derivee(Y, t):                     # Y d'abord, t ensuite : imposé par odeint
    theta, omega = Y                     # ordre 2 → système de 2 équations d'ordre 1
    return [omega, -g / L * np.sin(theta)]

t = np.linspace(0, 10, 2001)
for deg in [10, 90]:
    sol = odeint(derivee, [deg * np.pi / 180, 0], t)    # θ(0) en rad, ω(0) = 0
    plt.plot(t, sol[:, 0] * 180 / np.pi, label=str(deg) + '°')  # colonne 0 : θ

print('T0 =', round(2 * np.pi * np.sqrt(L / g), 2), 's')
plt.xlabel('t (s)')
plt.ylabel('θ (°)')
plt.legend()
plt.show()

T0 = 2.01 s

Sur le graphique : T ≈ 2,0 s à 10°, mais ≈ 2,4 s à 90°. Les grandes oscillations ne sont pas isochrones.
